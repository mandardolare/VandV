# Create a CAN message

                      data=[0x05, 0x62, 0x16, 0x56, 0x00, 0x4D,0xAA,0xAA],
                      is_extended_id=False)

       data_size = len(getdata)

         print("Received message on ID 0x1BDA00F1: {}".format(a.data))
